class Solution {
    public int secondLargestElement(int[] nums) {
        int fmax = -1;
        int smax = -1;

        for(int i=0; i<nums.length; i++)
        {
            if(nums[i] > fmax)
            {
                smax = fmax;
                fmax = nums[i];
                
            }
            if(nums[i] > smax && nums[i] < fmax)
            {
                smax = nums[i];
            }
        }
        return smax;
    }
}
